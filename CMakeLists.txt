cmake_minimum_required(VERSION 3.21)
project(PrintQueueORM LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SqliteOrm CONFIG REQUIRED)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
add_executable(PrintQueueORM src/main.cpp src/RawSql.cpp)
target_include_directories(PrintQueueORM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(PrintQueueORM PRIVATE sqlite_orm::sqlite_orm unofficial::sqlite3::sqlite3)
enable_testing()
add_test(NAME orm_demo COMMAND PrintQueueORM)
